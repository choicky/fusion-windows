cmake_minimum_required(VERSION 3.20)
project(FusionNativeFcitx LANGUAGES C CXX)
include(../common.cmake)
find_package(libuv CONFIG REQUIRED)
set(LIBUV_TARGET libuv::uv CACHE STRING "" FORCE)
add_subdirectory("${FUSION_UPSTREAM_DIR}/fcitx5" fcitx5)
target_link_libraries(Fcitx5Utils PRIVATE shell32 ole32)
find_package(getopt CONFIG REQUIRED)
target_link_libraries(Fcitx5Core PRIVATE getopt::getopt_shared)
target_sources(Fcitx5Utils PRIVATE "${FUSION_NATIVE_SUPPORT_DIR}/exports.cpp" "${FUSION_NATIVE_SUPPORT_DIR}/platform.cpp")
target_compile_definitions(Fcitx5Utils PRIVATE FUSION_PLATFORM_BUILD)
